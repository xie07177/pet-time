<script>
//初始化本地存储
if(!localStorage.getItem('petBase')) localStorage.setItem('petBase','{}');
if(!localStorage.getItem('healthRecords')) localStorage.setItem('healthRecords','[]');
if(!localStorage.getItem('communityPosts')) localStorage.setItem('communityPosts','[]');

//切换Tab
function switchTab(tabName){
    document.querySelectorAll('.tab-panel').forEach(el=>el.classList.add('hidden'));
    document.getElementById('tab-'+tabName).classList.remove('hidden');
    renderAll();
}

//保存宠物基础档案
function savePetBase(){
    const pet = {
        name:document.getElementById('petName').value,
        type:document.getElementById('petType').value,
        city:document.getElementById('petCity').value
    }
    localStorage.setItem('petBase', JSON.stringify(pet));
    alert("✅宠物档案保存成功");
    renderAll();
}

//新增健康记录
function addHealthRecord(){
    const type = document.getElementById('recordType').value;
    const val = document.getElementById('recordValue').value;
    const date = document.getElementById('recordDate').value;
    if(!val || !date){ alert('请填写完整内容'); return; }
    const arr = JSON.parse(localStorage.getItem('healthRecords'));
    arr.push({type,value:val,date});
    localStorage.setItem('healthRecords', JSON.stringify(arr));
    alert('✅健康记录已添加');
    renderAll();
}

//发布社区动态
function publishPost(){
    const content = document.getElementById('postContent').value.trim();
    if(!content) return alert('请输入动态内容');
    const petInfo = JSON.parse(localStorage.getItem('petBase'));
    const posts = JSON.parse(localStorage.getItem('communityPosts'));
    posts.push({
        content,
        petName: petInfo.name||"匿名宠主",
        city: petInfo.city||"未知城市",
        time: new Date().toLocaleString()
    })
    localStorage.setItem('communityPosts', JSON.stringify(posts));
    document.getElementById('postContent').value = '';
    renderAll();
}

//渲染全部数据
function renderAll(){
    const petBase = JSON.parse(localStorage.getItem('petBase'));
    const health = JSON.parse(localStorage.getItem('healthRecords'));
    const posts = JSON.parse(localStorage.getItem('communityPosts'));

    //回填表单
    document.getElementById('petName').value = petBase.name||'';
    document.getElementById('petType').value = petBase.type||'猫咪';
    document.getElementById('petCity').value = petBase.city||'';

    //渲染健康列表
    const hWrap = document.getElementById('healthList');
    if(health.length===0){
        hWrap.innerHTML = `<div class="text-gray-400">暂无记录，请添加第一条健康数据</div>`;
    }else{
        hWrap.innerHTML = health.map(item=>`
        <div class="border-b py-2 flex justify-between">
            <span>${item.date}</span>
            <span class="font-medium text-primary">${item.type}</span>
            <span>${item.value}</span>
        </div>`).join('');
    }

    //渲染时间线（按日期排序）
    const sorted = [...health].sort((a,b)=> new Date(a.date)-new Date(b.date));
    const tWrap = document.getElementById('timelineWrap');
    if(sorted.length===0){
        tWrap.innerHTML = `<div class="text-gray-400">还没有成长事件，录入健康记录就会生成时间线</div>`;
    }else{
        tWrap.innerHTML = sorted.map(i=>`
        <div class="relative">
            <div class="absolute -left-[28px] w-3 h-3 rounded-full bg-primary"></div>
            <div class="bg-soft p-3 rounded-lg">
                <div class="font-medium">${i.date}｜${i.type}</div>
                <div class="text-sm text-gray-600">${i.value}</div>
            </div>
        </div>`).join('');
    }

    //渲染社区动态流
    const pFeed = document.getElementById('postFeed');
    if(posts.length===0){
        pFeed.innerHTML = `<div class="text-gray-400">还没有动态，快去发布第一条养宠日常吧！</div>`;
    }else{
        pFeed.innerHTML = posts.reverse().map(item=>`
        <div class="border-b pb-3">
            <div class="text-sm text-gray-500">${item.petName} · ${item.city} · ${item.time}</div>
            <div class="mt-1">${item.content}</div>
        </div>`).join('');
    }

    //模拟同城宠友匹配
    const friendWrap = document.getElementById('cityFriendList');
    if(!petBase.city){
        friendWrap.innerHTML = `<div class="text-gray-400">请先完善你的宠物档案里【所在城市】</div>`;
    }else{
        //模拟同城用户
        const mockFriends = [
            {name:"喵小柒",pet:"布偶猫",city:petBase.city},
            {name:"养狗的阿泽",pet:"金毛犬",city:petBase.city},
            {name:"仓鼠团子妈",pet:"金丝熊",city:petBase.city}
        ]
        friendWrap.innerHTML = mockFriends.map(f=>`
        <div class="border p-2 rounded text-sm">
            <div class="font-medium">${f.name}</div>
            <div class="text-gray-500">饲养：${f.pet}</div>
            <div class="text-xs text-secondary">📍${f.city}</div>
        </div>`).join('');
    }
}

//页面加载渲染
window.onload = renderAll;
</script>
